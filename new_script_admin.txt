<script>
  /*
   * addCustomTags(gridId, inputId)
   * Textbox'taki değeri virgülle ayırır, her bir parçayı trim eder,
   * boş olmayanları tag-grid'e checked checkbox olarak ekler.
   * Aynı tag zaten varsa tekrar eklemez.
   */
  function addCustomTags(gridId, inputId) {
    const input = document.getElementById(inputId);
    const grid  = document.getElementById(gridId);
    const raw   = input.value.trim();
    if (!raw) return;

    const tags = raw.split(',').map(t => t.trim()).filter(t => t.length > 0);

    tags.forEach(tagText => {
      // Aynı isimde tag zaten var mı kontrol et
      const existing = Array.from(grid.querySelectorAll('.tag-check'));
      const duplicate = existing.some(el => el.textContent.trim().toLowerCase() === tagText.toLowerCase());
      if (duplicate) return;

      const label = document.createElement('label');
      label.className = 'tag-check custom-tag';
      label.innerHTML = `<input type="checkbox" checked> ${tagText}`;
      grid.appendChild(label);
    });

    input.value = '';
  }

  // Enter tuşuyla da tag eklenebilsin
  document.addEventListener('keydown', function(e) {
    if (e.key === 'Enter') {
      const active = document.activeElement;
      if (active && active.id === 'bCustomTag') addCustomTags('bTagGrid', 'bCustomTag');
      if (active && active.id === 'dCustomTag') addCustomTags('dTagGrid', 'dCustomTag');
    }
  });

  // Load resources on page load
  document.addEventListener('DOMContentLoaded', loadResources);

  function loadResources() {
    const resourcesTableBody = document.getElementById('resourcesTableBody');

    // Show loading state
    const loadingMessage = document.createElement('tr');
    loadingMessage.innerHTML = '<td colspan="5">Kaynaklar yükleniyor...</td>';
    loadingMessage.className = 'loading';
    if (resourcesTableBody) {
      resourcesTableBody.innerHTML = '';
      resourcesTableBody.appendChild(loadingMessage);
    }

    // Fetch resources from API
    fetch('/api/resources')
      .then(response => response.json())
      .then(data => {
        if (data.success) {
          displayResources(data.resources);
        } else {
          showError('Kaynaklar yüklenirken bir hata oluştu.');
        }
      })
      .catch(error => {
        showError('Sunucuya bağlanılamadı. Lütfen daha sonra tekrar deneyin.');
        console.error('Error:', error);
      });
  }

  function displayResources(resources) {
    const resourcesTableBody = document.getElementById('resourcesTableBody');
    if (!resourcesTableBody) return;

    resourcesTableBody.innerHTML = '';

    if (resources.length === 0) {
      const noDataRow = document.createElement('tr');
      noDataRow.innerHTML = '<td colspan="5">Henüz kaynak eklenmedi.</td>';
      resourcesTableBody.appendChild(noDataRow);
      return;
    }

    resources.forEach(resource => {
      const row = document.createElement('tr');
      row.innerHTML = `
        <td><strong>${resource.title}</strong></td>
        <td>${resource.author || '-'}</td>
        <td>${resource.tags || '-'}</td>
        <td><span class="status-badge ${resource.is_available ? 'status-active' : 'status-checked'}">
            ${resource.is_available ? 'Aktif' : 'Ödünç Alındı'}
          </span></td>
        <td>
          ${resource.file_path ? `<button class="btn-disa" onclick="downloadResource('${resource.file_path}')">Dışa Aktar</button>` : ''}
        </td>
      `;
      resourcesTableBody.appendChild(row);
    });
  }

  function showError(message) {
    const resourcesTableBody = document.getElementById('resourcesTableBody');
    if (resourcesTableBody) {
      const errorRow = document.createElement('tr');
      errorRow.innerHTML = `<td colspan="5" style="color: #991b1b;">${message}</td>`;
      resourcesTableBody.innerHTML = '';
      resourcesTableBody.appendChild(errorRow);
    }
  }

  // Enhance the search functionality
  function doSearch() {
    const searchInput = document.getElementById('searchInput');
    const query = searchInput.value.trim();

    if (!query) {
      loadResources(); // Load all if empty
      return;
    }

    // Show loading
    const resourcesTableBody = document.getElementById('resourcesTableBody');
    if (resourcesTableBody) {
      resourcesTableBody.innerHTML = '<tr><td colspan="5">Arama yapılıyor...</td></tr>';
    }

    // Search via API
    fetch(`/api/resources/search?q=${encodeURIComponent(query)}`)
      .then(response => response.json())
      .then(data => {
        if (data.success) {
          displayResources(data.resources);
        } else {
          showError('Arama sırasında bir hata oluştu.');
        }
      })
      .catch(error => {
        showError('Sunucuya bağlanılamadı.');
        console.error('Error:', error);
      });
  }

  // Update the existing doSearch function to use AJAX
  window.doSearch = doSearch;

  // Also handle Enter key in search
  document.getElementById('searchInput').addEventListener('keypress', function(e) {
    if (e.key === 'Enter') {
      doSearch();
    }
  });

  // AJAX functions for admin operations

  function uploadStudents() {
    const f = document.getElementById('studentFile').files[0];
    if (!f) { alert('Lütfen bir dosya seçin.'); return; }

    const formData = new FormData();
    formData.append('file', f);

    fetch('/api/students/upload', {
      method: 'POST',
      body: formData
    })
    .then(response => response.json())
    .then(data => {
      if (data.success) {
        alert('Öğrenci listesi yüklendi: ' + data.added + ' öğrenci eklendi, ' + data.updated + ' öğrenci güncellendi.');
        document.getElementById('studentFile').value = '';
      } else {
        alert('Yükleme başarısız: ' + data.message);
      }
    })
    .catch(error => {
      alert('Bir hata oluştu: ' + error.message);
      console.error('Error:', error);
    });
  }

  function exportDB() {
    fetch('/api/admin/export')
      .then(response => {
        if (response.ok) {
          return response.blob();
        } else {
          return response.json().then(err => { throw new Error(err.message); });
        }
      })
      .then(blob => {
        const url = window.URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.style.display = 'none';
        a.href = url;
        a.download = 'library.db';
        document.body.appendChild(a);
        a.click();
        window.URL.revokeObjectURL(url);
        alert('Veri tabanı başarıyla dışa aktarıldı.');
      })
      .catch(error => {
        alert('Dışa aktarma başarısız: ' + error.message);
        console.error('Error:', error);
      });
  }

  function importDB() {
    const f = document.getElementById('importFile').files[0];
    if (!f) { alert('Lütfen bir .db dosyası seçin.'); return; }
    if (!f.name.endsWith('.db')) { alert('Lütfen bir .db dosyası seçin.'); return; }

    const formData = new FormData();
    formData.append('file', f);

    fetch('/api/admin/import', {
      method: 'POST',
      body: formData
    })
    .then(response => response.json())
    .then(data => {
      if (data.success) {
        alert('Veri tabanı başarıyla yüklendi. Lütfen uygulamayı yeniden başlatın.');
        document.getElementById('importFile').value = '';
      } else {
        alert('Yükleme başarısız: ' + data.message);
      }
    })
    .catch(error => {
      alert('Bir hata oluştu: ' + error.message);
      console.error('Error:', error);
    });
  }

  function saveBasili() {
    const ad = document.getElementById('bAd').value.trim();
    const yazar = document.getElementById('bYazar').value.trim();
    const dil = document.querySelector('input[name="bDil"]:checked').value;
    const raf = document.getElementById('bRaf').value.trim();

    if (!ad || !yazar || !raf) {
      alert('Lütfen zorunlu alanları doldurun.');
      return;
    }

    // Process tags
    const tagGrid = document.getElementById('bTagGrid');
    const tags = Array.from(tagGrid.querySelectorAll('.tag-check input:checked'))
                      .map(cb => cb.parentNode.textContent.trim());
    const customTagInput = document.getElementById('bCustomTag');
    const customTags = customTagInput.value.trim().split(',').map(t => t.trim()).filter(t => t.length > 0);
    const allTags = [...tags, ...customTags].join(',');

    const data = {
      title: ad,
      author: yazar,
      language: dil,
      shelf_location: raf,
      tags: allTags,
      resource_type: 'physical'
    };

    fetch('/api/resources/physical', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(data)
    })
    .then(response => response.json())
    .then(data => {
      if (data.success) {
        alert('Basılı kaynak kaydedildi: ' + ad);
        // Clear form
        document.getElementById('bAd').value = '';
        document.getElementById('bYazar').value = '';
        document.getElementById('bRaf').value = '';
        document.getElementById('bCustomTag').value = '';
        // Uncheck all tags
        const checks = document.getElementById('bTagGrid').querySelectorAll('.tag-check input');
        checks.forEach(cb => cb.checked = false);
        // Reload resources
        loadResources();
      } else {
        alert('Kaydetme başarısız: ' + data.message);
      }
    })
    .catch(error => {
      alert('Bir hata oluştu: ' + error.message);
      console.error('Error:', error);
    });
  }

  function showFiles(input) {
    const list  = document.getElementById('fileList');
    const names = document.getElementById('uploadedNames');
    list.innerHTML = '';
    let nameList = [];
    for (const f of input.files) {
      const li = document.createElement('li');
      li.innerHTML = '📄 ' + f.name;
      list.appendChild(li);
      nameList.push(f.name);
    }
    names.textContent = input.files.length + ' dosya seçildi: ' + nameList.join(', ');
  }

  function saveDijital() {
    const f = document.getElementById('dijFile').files;
    if (!f.length) { alert('Lütfen en az bir dosya seçin.'); return; }

    // We'll handle the first file for simplicity (multiple file upload not implemented in API yet)
    const file = f[0];
    const formData = new FormData();
    formData.append('file', file);

    // Get other metadata from form
    const titel = document.getElementById('dAd')?.value.trim() || '';
    const yazar = document.getElementById('dYazar')?.value.trim() || '';
    const dil = document.querySelector('input[name="dDil"]:checked')?.value || '';
    const raf = document.getElementById('dRaf')?.value.trim() || '';
    const tagGrid = document.getElementById('dTagGrid');
    const tags = Array.from(tagGrid.querySelectorAll('.tag-check input:checked'))
                      .map(cb => cb.parentNode.textContent.trim());
    const customTagInput = document.getElementById('dCustomTag');
    const customTags = customTagInput.value.trim().split(',').map(t => t.trim()).filter(t => t.length > 0);
    const allTags = [...tags, ...customTags].join(',');

    // Append metadata to formData
    formData.append('title', titel);
    formData.append('author', yazar);
    formData.append('language', dil);
    formData.append('shelf_location', raf);
    formData.append('tags', allTags);
    formData.append('resource_type', 'digital');

    fetch('/api/resources/digital', {
      method: 'POST',
      body: formData
    })
    .then(response => response.json())
    .then(data => {
      if (data.success) {
        alert(f.length + ' dijital kaynak kaydedildi.');
        // Clear form
        document.getElementById('dAd').value = '';
        document.getElementById('dYazar').value = '';
        document.getElementById('dRaf').value = '';
        document.getElementById('dCustomTag').value = '';
        document.getElementById('dijFile').value = '';
        document.getElementById('uploadedNames').textContent = 'Henüz dosya seçilmedi.';
        document.getElementById('fileList').innerHTML = '';
        // Uncheck all tags
        const checks = document.getElementById('dTagGrid').querySelectorAll('.tag-check input');
        checks.forEach(cb => cb.checked = false);
        // Reload resources
        loadResources();
      } else {
        alert('Kaydetme başarısız: ' + data.message);
      }
    })
    .catch(error => {
      alert('Bir hata oluztu: ' + error.message);
      console.error('Error:', error);
    });
  }

  // Function to download a digital resource (placeholder)
  function downloadResource(filePath) {
    // In a real implementation, we would serve the file from the uploads directory
    // For now, we'll just show an alert
    alert('Dosya indirme işlemi şu anda desteklenmiyor. Dosya yolu: ' + filePath);
  }
</script>
